def angle_to_vector(angle):
    return Math.Cos(angle), Math.Sin(angle)

def euler_to_vector(euler):
    yaw, pitch, roll = euler

    # First calculate the "fan pattern" ignoring pitch

    print(yaw, pitch, roll)

    pitch = pitch % (2*math.pi)
    yaw = yaw % (2*math.pi)
    roll = roll % (2*math.pi)

    x = math.sin(yaw)
    z = math.cos(yaw)


    # First formula finds x, y, and z from angles
    # AHA! the 


    # Then offset that using the pitch

    # Look at me using SOHCAHTOA
    # Are you proud of me algebra 2 teacher?

    # We have the length of the hypot (just 1, 1 unit ahead)
    # and the angle, SOH and CAH will be used to find the lengths of the other sides.

    # Solve for opp (y axis)
    # sin(theta) = opp / hypot
    # opp = sin(pitch)*hypot
    y = math.sin(pitch)

    # Now for CAH
    # cos(th) = adj / hypot
    # adj = cos(pitch)*hypot
    z += math.cos(pitch)

    return x, y, z
